############################################################
# 1. Calculate geographic distance matrix for IBD
############################################################

# Install and load the required package
install.packages("geosphere")
library(geosphere)

# Read the coordinate file
# The file should contain at least three columns: Name, Longitude, and Latitude
data <- read.csv("H://2025//ME//GYLW.csv")

# Extract longitude and latitude
coordinates <- data[, c("Longitude", "Latitude")]

# Calculate geographic distance matrix in kilometres
distance_matrix <- as.matrix(distm(coordinates) / 1000)

# Use population/site names as row and column names
row_col_names <- as.character(data$Name)

distance_df <- data.frame(distance_matrix)
rownames(distance_df) <- row_col_names
colnames(distance_df) <- row_col_names

# Save the geographic distance matrix
write.csv(distance_df, "H://2025//ME//distance_matrixnew.csv", row.names = TRUE)



############################################################
# 2. Generate standardized pairwise environmental-distance
#    matrices for IBE
############################################################

# Install and load required packages
install.packages("readxl")
install.packages("writexl")

library(readxl)
library(writexl)

# Set working directory
setwd("H://2025//ME//new1//IBE//")

# Read environmental data
# The first column should be Name, and the remaining columns should be environmental variables
data <- read_excel("H://2025//ME//new1//IBE//rawdata.xlsx")

# Extract site names and environmental-variable names
names <- as.character(data$Name)
value_columns <- colnames(data)[-1]

# Create an output folder for pairwise matrices
output_folder <- "pairwise_matricesscale"
dir.create(output_folder, showWarnings = FALSE)

# For each environmental variable:
# 1. standardize the variable;
# 2. calculate pairwise absolute differences;
# 3. save the matrix as a CSV file.
for (col in value_columns) {

  values <- data[[col]]

  # Standardize the variable to mean = 0 and SD = 1
  scaled_values <- as.vector(scale(values))

  # Calculate pairwise absolute differences
  diff_matrix <- abs(outer(scaled_values, scaled_values, "-"))

  # Convert the matrix to a data frame
  diff_df <- as.data.frame(diff_matrix)
  rownames(diff_df) <- names
  colnames(diff_df) <- names

  # Add site names as the first column
  diff_df <- cbind("NAME" = names, diff_df)

  # Save the pairwise matrix
  output_file <- file.path(output_folder, paste0(col, "_pairwise_scaled.csv"))
  write.csv(diff_df, file = output_file, row.names = FALSE)
}

print(paste("All standardized pairwise matrices were saved to:", output_folder))
